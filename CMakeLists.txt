cmake_minimum_required(VERSION 3.24)
project(smartknob_core C CXX)
set(CMAKE_C_STANDARD 11)
option(SK_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(SK_SANITIZE AND NOT EMSCRIPTEN)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(sk_core core/src/project.c core/src/haptics.c core/src/leds.c core/src/protocol.c core/src/encoder.c)
target_include_directories(sk_core PUBLIC core/include)
target_compile_options(sk_core PRIVATE -Wall -Wextra -Werror)
target_link_libraries(sk_core PUBLIC m)
enable_testing()
add_executable(core_tests tests/core_tests.c)
target_link_libraries(core_tests PRIVATE sk_core)
add_test(NAME core COMMAND core_tests)
if(NOT EMSCRIPTEN)
  add_executable(device_emulator tests/device_emulator.c)
  target_link_libraries(device_emulator PRIVATE sk_core)
  add_executable(fuzz_bundle tests/fuzz_bundle.c)
  target_link_libraries(fuzz_bundle PRIVATE sk_core)
endif()
add_executable(bundle_check tests/bundle_check.c)
target_link_libraries(bundle_check PRIVATE sk_core)
if(LVGL_DIR)
  set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/preview/lv_conf.h" CACHE PATH "" FORCE)
  set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
  set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
  add_subdirectory(${LVGL_DIR} ${CMAKE_BINARY_DIR}/lvgl)
  add_library(sk_render core/src/render.c)
  target_include_directories(sk_render PUBLIC core/include)
  target_link_libraries(sk_render PUBLIC sk_core lvgl)
  if(EMSCRIPTEN)
    add_executable(preview preview/preview.c)
    target_link_libraries(preview PRIVATE sk_render)
    set_target_properties(preview PROPERTIES SUFFIX ".js")
    target_link_options(preview PRIVATE "--no-entry" "-sMODULARIZE=1" "-sEXPORT_ES6=1" "-sEXPORT_NAME=createPreview" "-sALLOW_MEMORY_GROWTH=1" "-sENVIRONMENT=web,node" "-sEXPORTED_RUNTIME_METHODS=['HEAPU8','HEAPF32']" "-sEXPORTED_FUNCTIONS=['_malloc','_free','_sk_preview_init','_sk_preview_load','_sk_preview_select','_sk_preview_step','_sk_preview_render','_sk_preview_leds','_sk_preview_reset']")
  else()
    add_executable(render_check tests/render_check.c preview/preview.c)
    target_link_libraries(render_check PRIVATE sk_render)
    add_test(NAME render COMMAND render_check)
  endif()
endif()
